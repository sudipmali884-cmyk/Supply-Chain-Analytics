
import random
import numpy as np
import pandas as pd
from faker import Faker

fake = Faker("en_IN")

random.seed(42)
np.random.seed(42)
Faker.seed(42)

OUTPUT_DIR = "../data"
os.makedirs(OUTPUT_DIR, exist_ok=True)

TOTAL_PRODUCTS = 50_000

categories = {
    "Electronics": [
        "Computer Accessories",
        "Mobile Accessories",
        "Networking",
        "Audio",
        "Storage"
    ],
    "Industrial": [
        "Safety Equipment",
        "Tools",
        "Machinery Parts",
        "Electrical Components"
    ],
    "Office": [
        "Stationery",
        "Office Equipment",
        "Furniture"
    ],
    "Packaging": [
        "Boxes",
        "Plastic Packaging",
        "Labels",
        "Wrapping Material"
    ],
    "Automotive": [
        "Engine Parts",
        "Brake Components",
        "Electrical Parts",
        "Accessories"
    ]
}

brands = [
    "Bosch",
    "Tata",
    "LG",
    "Samsung",
    "Dell",
    "HP",
    "Lenovo",
    "SKF",
    "Siemens",
    "Philips",
    "Havells",
    "Bajaj",
    "Godrej",
    "Voltas",
    "Generic"
]

units = [
    "Piece",
    "Box",
    "Kg",
    "Liter",
    "Meter",
    "Pack",
    "Set"
]

statuses = [
    "Active",
    "Active",
    "Active",
    "Active",
    "Inactive",
    "Discontinued"
]

rows = []

for i in range(1, TOTAL_PRODUCTS + 1):

    category = random.choice(list(categories.keys()))
    sub_category = random.choice(categories[category])
    brand = random.choice(brands)

    unit_cost = round(random.uniform(50, 50000), 2)

    markup = random.uniform(1.08, 1.60)

    selling_price = round(unit_cost * markup, 2)

    reorder_level = random.randint(10, 1000)

    rows.append({
        "product_id": f"PRD{i:06d}",
        "product_name": f"{brand} {sub_category} {random.randint(100, 9999)}",
        "category": category,
        "sub_category": sub_category,
        "brand": brand,
        "unit_cost": unit_cost,
        "selling_price": selling_price,
        "reorder_level": reorder_level,
        "unit_of_measure": random.choice(units),
        "product_status": random.choice(statuses)
    })


df = pd.DataFrame(rows)

# ---------------------------------------------------------
# DIRTY DATA
# ---------------------------------------------------------

# 1. Product name whitespace
idx = np.random.choice(df.index, 700, replace=False)

df.loc[idx, "product_name"] = (
    df.loc[idx, "product_name"]
    .astype(str)
    .apply(lambda x: random.choice([
        f" {x}",
        f"{x} ",
        f"  {x}  ",
        x.replace(" ", "  ")
    ]))
)

# 2. Brand casing problems
idx = np.random.choice(df.index, 500, replace=False)

df.loc[idx, "brand"] = (
    df.loc[idx, "brand"]
    .astype(str)
    .apply(lambda x: random.choice([
        x.lower(),
        x.upper(),
        f" {x}",
        f"{x} "
    ]))
)

# 3. Missing reorder levels
idx = np.random.choice(df.index, 250, replace=False)
df.loc[idx, "reorder_level"] = np.nan

# 4. Invalid unit values
idx = np.random.choice(df.index, 150, replace=False)

dirty_units = [
    "unknown",
    "N/A",
    "unit",
    "pieces",
    "",
    " KG ",
    None
]

df.loc[idx, "unit_of_measure"] = [
    random.choice(dirty_units)
    for _ in idx
]

# 5. Invalid prices
idx = np.random.choice(df.index, 120, replace=False)

df.loc[idx, "unit_cost"] = [
    random.choice([
        -50,
        0,
        np.nan,
        999999.99
    ])
    for _ in idx
]

# 6. Dirty status
idx = np.random.choice(df.index, 200, replace=False)

dirty_statuses = [
    "active",
    "ACTIVE",
    " Active",
    "Active ",
    "unknown",
    "N/A",
    None
]

df.loc[idx, "product_status"] = [
    random.choice(dirty_statuses)
    for _ in idx
]

# ---------------------------------------------------------
# SAVE
# ---------------------------------------------------------

file_path = os.path.join(OUTPUT_DIR, "products_master.csv")

df.to_csv(
    file_path,
    index=False
)

print("=" * 60)
print("PRODUCT MASTER GENERATED")

print(f"Rows       : {len(df):,}")
